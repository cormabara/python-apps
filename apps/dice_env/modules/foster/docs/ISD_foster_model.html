Foster model implementation for ISD2 power modules temperature junction estimation

NBEL, 25:05/22
clear;
close all;
SIM_LEN = 1000;
Input data are the COMPARE values of the SVPVM and the actual phase currents:
%data file recover from test-bench
%data_filename = 'oscillo';
%VarNum = 1;
CompareU = ones(1,SIM_LEN);
%DICE_data_import(data_filename,CompareU, VarNum, SIM_LEN);
CMP_RATE = 5000;
ActCurrU = ones(1,SIM_LEN); %[A] ------------>dati sperimentali
%VarNum = 2;
%DICE_data_import(data_filename,CompareU, VarNum, SIM_LEN);
Referring to the semikron IPM:
VCE_IGBT = 2.45; %[V]
FCHOP = 5e3; % [Hz] PWm frequency
PswitchIgbt = 14.1e-3+26.4e-3*FCHOP; %[W] (Eon+Eoff)*f @225A, 600V
PswitchDiode =17e-3*FCHOP; %[W] (Erec)*f @225A
VD = 2.42; %[V] @250A dc
%foster model
RTH_IGBT = 0.14; %[K/W]
RTH_DIODE = 0.2; %[K/W]
Tau_IGBT = 0.04; %[s]
Tau_DIODE = 0.04; %[s]
% tuning parametri foster model
tt=0.001:0.001:5;
OutIGBT=RTH_IGBT*(1-exp(-tt/Tau_IGBT));
OutDIODE=RTH_DIODE*(1-exp(-tt/Tau_DIODE));
figure;
loglog(tt,[OutIGBT; OutDIODE]);
legend('IGBT', 'DIODE');
grid on;
%%
%--->inizializzazione filtri passabasso
Fc = 1/1e-3; %[s] Sampling 1ms
%banda passante
IGBT_LP_Ft = 1/(Tau_IGBT*2*pi); %[Hz]
DIODE_LP_Ft = 1/(Tau_DIODE*2*pi); %[Hz]
% Ft=(wt*Fc)/(2*pi*2^16);
IGBT_wt= (IGBT_LP_Ft*2*pi*2^16)/Fc;
DIODE_wt= (DIODE_LP_Ft*2*pi*2^16)/Fc;
%variabili locali
PdissIgbtTopACC=0;
PdissIgbtBotACC=0;
PdissDiodeTopACC = 0;
PdissDiodeBotACC = 0;
tt = zeros(1,SIM_LEN);
DBG_PdissIgbtTop = zeros(1,SIM_LEN);
DBG_PdissIgbtBot =zeros(1,SIM_LEN);
DBG_PdissDiodeTop =zeros(1,SIM_LEN);
DBG_PdissDiodeBot =zeros(1,SIM_LEN);
DBG_TempIgbtTop= zeros(1,SIM_LEN);
DBG_TempIgbtBop= zeros(1,SIM_LEN);
DBG_TempDiodeTop= zeros(1,SIM_LEN);
DBG_TempDiodeBop= zeros(1,SIM_LEN);
%--------------->il ciclo for simula l'IRQ
for i=1:SIM_LEN
PdissIgbtTop = 0;
PdissIgbtBot = 0;
PdissDiodeTop = 0;
PdissDiodeBot = 0;
foster model: we'll just implement one time constant cause of sw resources limitations:
if (ActCurrU(i)>0)
PdissIgbtTop = ActCurrU(i)*VCE_IGBT*CompareU(i)/CMP_RATE+PswitchIgbt;
PdissDiodeBot = ActCurrU(i)*VD*(1-CompareU(i)/CMP_RATE)+PswitchDiode;
else
PdissIgbtBot = ActCurrU(i)*VCE_IGBT*(1-CompareU(i)/CMP_RATE)+PswitchIgbt;
PdissDiodeTop = ActCurrU(i)*VD*CompareU(i)/CMP_RATE+PswitchDiode;
end
%filtro per stima temperatura
% Ft=(wt*Fc)/(2*pi*2^16);
% err = powerpulse - mem/2^16;
% mem = mem + err*wt;
% out = mem/2^16;
%top IGBT
err = PdissIgbtTop - PdissIgbtTopACC/2^16;
PdissIgbtTopACC = PdissIgbtTopACC + err*IGBT_wt;
TempIgbtTop = RTH_IGBT * PdissIgbtTopACC/2^16;
%bot IGBT
err = PdissIgbtBot - PdissIgbtBotACC/2^16;
PdissIgbtBotACC = PdissIgbtBotACC + err*IGBT_wt;
TempIgbtBot = RTH_IGBT * PdissIgbtBotACC/2^16;
%top DIODE
err = PdissDiodeTop - PdissDiodeTopACC/2^16;
PdissDiodeTopACC = PdissDiodeTopACC + err*IGBT_wt;
TempDiodeTop = RTH_IGBT * PdissDiodeTopACC/2^16;
%bot DIODE
err = PdissDiodeBot - PdissDiodeBotACC/2^16;
PdissDiodeBotACC = PdissDiodeBotACC + err*IGBT_wt;
TempDiodeBot = RTH_IGBT * PdissDiodeBotACC/2^16;
%---->DEBUG
tt(i) = i/Fc;
DBG_PdissIgbtTop(i) = PdissIgbtTop;
DBG_PdissIgbtBot(i) = PdissIgbtBot;
DBG_PdissDiodeTop(i) = PdissDiodeTop;
DBG_PdissDiodeBot(i) = PdissDiodeBot;
DBG_TempIgbtTop(i)= TempIgbtTop;
DBG_TempIgbtBot(i)= TempIgbtBot;
DBG_TempDiodeTop(i)= TempDiodeTop;
DBG_TempDiodeBot(i)= TempDiodeBot;
end
figure;
plot(tt, [DBG_PdissIgbtTop; DBG_TempIgbtTop; DBG_PdissIgbtBot; DBG_TempIgbtBot]);
grid on;
legend('Pdiss IGBT top', 'Temp IGBT top', 'Pdiss IGBT bot', 'Temp IGBT bot');
figure;
plot(tt, [DBG_PdissDiodeTop; DBG_TempDiodeTop; DBG_PdissDiodeBot; DBG_TempDiodeBot]);
grid on;
legend('Pdiss Diode top', 'Temp Diode top', 'Pdiss Diode bot', 'Temp Diode bot');
